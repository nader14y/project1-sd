۱. اصلاً Testing چیست؟

فرض کن این تابع را نوشته‌ای:

def add(a, b):
    return a + b

برای اینکه مطمئن شوی خروجی درست است، می‌توانی دستی چاپش کنی:

print(add(2, 3))

اما اگر پروژه ۵۰۰ تابع داشته باشد، این روش دیگر قابل اتکا نیست. تست خودکار یعنی انتظارمان را به صورت کد بنویسیم:

assert add(2, 3) == 5
معنی assert: من انتظار دارم عبارت سمت راست درست باشد؛ اگر نیست، تست باید Fail شود.

pytest چیست؟

pytest یک فریم‌ورک تست برای Python است که تست‌ها را پیدا می‌کند، اجرا می‌کند و نتیجه را به صورت Pass/Fail گزارش می‌دهد.

def add(a, b):
    return a + b

def test_add():
    assert add(2, 3) == 5

۵. تست Exception و Float

تست Exception

def divide(a, b):
    if b == 0:
        raise ValueError("Cannot divide by zero")
    return a / b
import pytest

def test_divide_by_zero():
    with pytest.raises(ValueError):
        divide(10, 0)

بررسی متن خطا:

def test_divide_by_zero():
    with pytest.raises(
        ValueError,
        match="Cannot divide by zero"
    ):
        divide(10, 0)

مقایسه اعداد اعشاری

import pytest

def test_float():
    assert 0.1 + 0.2 == pytest.approx(0.3)
برای Floatها از pytest.approx استفاده کن تا خطاهای طبیعی precision در محاسبات باینری دردسرساز نشوند.

۶. Fixture — یکی از مهم‌ترین مفاهیم pytest

Fixture داده یا dependency مشترک تست‌ها را آماده می‌کند.

import pytest

@pytest.fixture
def user():
    return {
        "name": "Ali",
        "age": 30,
        "active": True,
    }

استفاده:

def test_user_name(user):
    assert user["name"] == "Ali"

def test_user_age(user):
    assert user["age"] == 30
pytest نام پارامتر تست را می‌بیند، Fixture هم‌نام را پیدا می‌کند، آن را اجرا می‌کند و مقدارش را Inject می‌کند.

Fixture به عنوان Dependency Injection

@pytest.fixture
def user():
    return {"name": "Ali"}

@pytest.fixture
def order(user):
    return {
        "id": 1,
        "user": user,
        "price": 200
    }

def test_order_user(order):
    assert order["user"]["name"] == "Ali"

Fixture با yield

@pytest.fixture
def database():
    print("Connect database")

    db = "connection"

    yield db

    print("Close database")

قبل از yield = Setup، بعد از yield = Teardown.

Fixture Scope

Scopeزمان ایجاد
functionبرای هر تست
classبرای هر کلاس تست
moduleبرای هر فایل
packageبرای هر پکیج
sessionیک بار برای کل اجرای pytest

Factory Fixture

@pytest.fixture
def make_user():
    def _make_user(name, age):
        return {
            "name": name,
            "age": age
        }

    return _make_user

def test_users(make_user):
    ali = make_user("Ali", 30)
    sara = make_user("Sara", 25)

    assert ali["name"] == "Ali"
    assert sara["name"] == "Sara"

۸. Parametrize

وقتی یک رفتار را با چند ورودی می‌خواهی تست کنی، به جای چند تست تکراری از Parametrize استفاده کن.

import pytest

@pytest.mark.parametrize(
    "a,b,expected",
    [
        (1, 2, 3),
        (2, 3, 5),
        (10, 20, 30),
        (-1, 1, 0),
        (0, 0, 0),
    ]
)
def test_add(a, b, expected):
    assert add(a, b) == expected

        pytest.param(18, True, id="minimum-adult"),
        pytest.param(30, True, id="adult"),
    ]
)
def test_is_adult(age, expected):
    ...

۱۱. Mock و Monkeypatch

Monkeypatch برای Environment Variable

import os

def get_environment():
    return os.getenv("APP_ENV")

def test_environment(monkeypatch):
    monkeypatch.setenv(
        "APP_ENV",
        "test"
    )

    assert get_environment() == "test"

Monkeypatch برای Function

# service.py

def get_price_from_api():
    return 100

def calculate_total():
    return get_price_from_api() * 2
import service

def fake_price():
    return 50

def test_total(monkeypatch):
    monkeypatch.setattr(
        service,
        "get_price_from_api",
        fake_price
    )

    assert service.calculate_total() == 100

MagicMock

from unittest.mock import MagicMock

def test_payment():
    payment_gateway = MagicMock()

    payment_gateway.charge.return_value = True

    result = payment_gateway.charge(100)

    assert result is True
    payment_gateway.charge.assert_called_once_with(100)

مثال واقعی OrderService

class OrderService:

    def __init__(self, payment_gateway):
        self.payment_gateway = payment_gateway

    def checkout(self, amount):
        result = self.payment_gateway.charge(amount)

        if result:
            return "success"

        return "failed"
from unittest.mock import MagicMock

def test_checkout_success():
    gateway = MagicMock()
    gateway.charge.return_value = True

    service = OrderService(gateway)

    result = service.checkout(200)

    assert result == "success"
    gateway.charge.assert_called_once_with(200)

Side Effect

gateway.charge.side_effect = TimeoutError()
در Unit Test معمولاً API خارجی، Email، SMS، Payment Gateway، S3 و سرویس‌های بیرونی را Mock می‌کنیم.

user_service.py

class UserAlreadyExists(Exception):
    pass


class UserService:

    def __init__(self, repository, email_service):
        self.repository = repository
        self.email_service = email_service

    def register(self, email, name):
        if not email:
            raise ValueError("Email is required")

        if not name:
            raise ValueError("Name is required")

        existing = self.repository.find_by_email(email)

        if existing:
            raise UserAlreadyExists(email)

        user = {
            "email": email,
            "name": name,
            "active": True,
        }

        self.repository.save(user)
        self.email_service.send_welcome(email)

        return user

Happy Path

def test_register_user(
    user_service,
    repository,
    email_service,
):
    repository.find_by_email.return_value = None

    user = user_service.register(
        "ali@example.com",
        "Ali",
    )

    assert user == {
        "email": "ali@example.com",
        "name": "Ali",
        "active": True,
    }

    repository.save.assert_called_once_with(user)

    email_service.send_welcome.assert_called_once_with(
        "ali@example.com"
    )

Duplicate User

import pytest
from app.user_service import UserAlreadyExists

def test_duplicate_user(
    user_service,
    repository,
    email_service,
):
    repository.find_by_email.return_value = {
        "email": "ali@example.com"
    }

    with pytest.raises(UserAlreadyExists):
        user_service.register(
            "ali@example.com",
            "Ali",
        )

    repository.save.assert_not_called()
    email_service.send_welcome.assert_not_called()

Invalid Input با Parametrize

@pytest.mark.parametrize(
    "email,name,error",
    [
        ("", "Ali", "Email is required"),
        ("ali@example.com", "", "Name is required"),
        (None, "Ali", "Email is required"),
        ("ali@example.com", None, "Name is required"),
    ]
)
def test_invalid_registration(
    user_service,
    email,
    name,
    error,
):
    with pytest.raises(
        ValueError,
        match=error
    ):
        user_service.register(
            email,
            name,
        )

۱۶. ۱۰ تمرین از ساده تا حرفه‌ای

تمرین ۱ — جمع دو عدد: تابع add(a,b) بساز. ابتدا سه Test جدا و سپس نسخه Parametrize بنویس.
تمرین ۲ — تشخیص زوج: تابع is_even(number) را با ورودی‌های زوج و فرد تست کن.
تمرین ۳ — تقسیم و Exception: برای تقسیم بر صفر ValueError بده و Message را هم تست کن.
تمرین ۴ — Fixture کاربر: Fixtureای بساز که User با name، age و active تولید کند.
تمرین ۵ — Factory Fixture: با make_user چند User مختلف بساز.
تمرین ۶ — Temporary File: تابع ذخیره متن بساز و با tmp_path تستش کن.
تمرین ۷ — Environment Variable: با monkeypatch.setenv مقدار DATABASE_URL را تست کن.
تمرین ۸ — Mock Email: بعد از Register شدن User بررسی کن send_welcome() دقیقاً یک بار Call شده باشد.
تمرین ۹ — Order Service: Payment Gateway را Mock کن و حالت success / failed را تست کن.
تمرین ۱۰ — Mini Project: Register User با Repository، Email Service، Duplicate User، Invalid Input، Error Handling و Parametrize بساز.
