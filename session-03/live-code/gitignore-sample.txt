# محتوای فایل .gitignore برای جلسه‌ی ۲
#
# ⚠️ این فایل را به بچه‌ها ندهید — خودشان باید بسازند، و مهم‌تر:
#    خودشان باید بگویند چرا هر خط اینجاست.
#
# نحوه‌ی ساخت در VS Code:
#    راست‌کلیک روی پوشه‌ی پروژه → New File → اسم دقیقاً:  .gitignore
#    (با نقطه‌ی اول، بدون .txt — Notepad ویندوز پسوند .txt اضافه می‌کند!)
#
# محتوای امروز فقط دو خط است. عمداً کوتاه:

venv/
__pycache__/


# ─────────────────────────────────────────────────────────────────────
# چرا این دو خط؟ — بگذارید خودشان جواب بدهند، نگویید:
#
#   venv/          «سنگین است، و از روی requirements.txt دوباره ساخته می‌شود.»
#                  همان جمله‌ای که ۲۵ دقیقه پیش، در آزمون تعویض، خودشان کشف کردند.
#
#   __pycache__/   «پایتون خودش می‌سازدش؛ کدِ ما نیست.»
#                  اگر پرسیدند این چیست: «نسخه‌ی سریع‌شده‌ی کد شما برای اجرای بعدی.»
#
# 🌱 پیش‌آگهی که باید بلند بگویید:
#    «این فایل بعداً مهم‌تر هم می‌شود — وقتی پروژه داده‌ی کاربر و راز پیدا کند.»
#    (جلسه‌ی ۴: leaderboard.json  ·  جلسه‌ی ۱۰: dist/ و build/ و DSN)
#
# ⛔ امروز اضافه نکنید:
#    *.pyc  ·  .vscode/  ·  game.log  ·  dist/  ·  build/
#    هر خطی که دلیلش را امروز نمی‌فهمند، امروز جایش نیست.
